public class TestProject {

    public static void main(String[] args) {
        long start =  System.currentTimeMillis();
        int[] counter = {0};
        IntStream.range(0, 1000000).parallel().forEach(i->counter[0]++);
        long finish = System.currentTimeMillis();
        System.out.println("Получилось:" + counter[0]);
        System.out.println(" Итоговое время:" + (finish-start));
        
        
        long start1 =  System.currentTimeMillis();
        AtomicInteger counter1 = new AtomicInteger();
        IntStream.range(0, 1000000).parallel().forEach(i -> counter1.incrementAndGet());
        long finish1 = System.currentTimeMillis();
        System.out.println("Получилось:" + counter1.get());
        System.out.println(" Итоговое время:" + (finish1-start1));
        
        
        long start2 =  System.currentTimeMillis();
        long sum = IntStream.range(0, 1000000).parallel().count();
        long finish2 = System.currentTimeMillis();
        System.out.println("Получилось:" + sum);
        System.out.println(" Итоговое время:" + (finish2-start2));

        //Порядок 0-9
        long start3 =  System.currentTimeMillis();
        IntStream.range(0, 10).parallel().forEach(i -> System.out.print(i + ""));
        long finish3 = System.currentTimeMillis();
        System.out.println();
        System.out.println(" Итоговое время:" + (finish3-start3));
        
        long start4 =  System.currentTimeMillis();
        List<Integer> result = IntStream.range(0, 10).parallel().boxed().collect(Collectors.toList());
        long finish4 = System.currentTimeMillis();
        System.out.println(result);
        System.out.println(" Итоговое время:" + (finish4-start4));
        System.out.println();
        
        
        long start5 =  System.currentTimeMillis();
        List<Integer> evens = IntStream.range(0, 100).parallel().filter(n -> n % 2 == 0).boxed().collect(Collectors.toList());
        long finish5 = System.currentTimeMillis();
        System.out.println("Четные:");
        System.out.println(evens);
        System.out.println(" Итоговое время:" + (finish5-start5));
    }
}
